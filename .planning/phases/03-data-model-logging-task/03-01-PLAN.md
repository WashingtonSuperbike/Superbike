```cpp
static void decipherCellsVoltage(twai_message_t msg, DashboardBatteryVoltages *battery)
{
    if (msg.data_length_code < 8) { ... return; }
    static float hv_cell_voltages[CONFIG_HV_CELL_COUNT] = {};
    uint32_t msgID  = msg.identifier;
    int cellOffset  = (((msgID >> 8) & 0xF) - 0x9);
    int ltcOffset   = (msgID & 0x1);
    int totalOffset = (cellOffset * 4) + (ltcOffset * 12);
    for (int i = 0; i < 4; i++) {
        uint16_t *buf = (uint16_t *)msg.data;
        hv_cell_voltages[i + totalOffset] = ((float)buf[i]) / 10000.0f;
    }
    float sum = 0.0f;
    for (int i = 0; i < CONFIG_HV_CELL_COUNT; i++) {
        sum += hv_cell_voltages[i];
    }
    battery->hv_series_voltage = sum;
}
```

```cpp
bool sd_open_log_file(const char *filename) {
    if (s_log_file.isOpen()) {
        s_log_file.close();
    }
    return s_log_file.open(filename, O_WRONLY | O_CREAT | O_TRUNC);
}

bool sd_write_log_line(const char *buf, size_t len) {
    if (!s_log_file.isOpen()) return false;
    return (size_t)s_log_file.write(buf, len) == len;
}

bool sd_sync_log_file() {
    if (!s_log_file.isOpen()) return false;
    return s_log_file.sync();
}

void sd_close_log_file() {

        s_log_file.sync();
        s_log_file.close();
    }
}
```

| T-03-02 | I (Info Disclosure) | hv_cell_voltages array | accept | Cell voltages are not PII; CSV on SD card is physically accessible to bike owner only. |
